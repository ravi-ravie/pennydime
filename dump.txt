import React, { useState, useEffect, useMemo } from 'react';
import Sidebar from './layouts/Sidebar';
import Header from './layouts/Header';
import MobileNav from './layouts/MobileNav';
import AddEntryModal from './components/common/AddEntryModal';
import AuthPage from './pages/auth/AuthPage';
import AppRoutes, { VALID_ROUTES } from './routes/AppRoutes';
import SetStartingModal from './components/dashboard/SetStartingModal';
import ReconcileModal from './components/dashboard/ReconcileModal';
import AdjustPoolModal from './components/dashboard/AdjustPoolModal';
import AdjustBudgetModal from './components/budget/AdjustBudgetModal';
import { initialRealData, defaultCategories, sampleDemoTransactions } from './data/initialData';

export default function App() {
  // 1. Authentication State
  const [currentUser, setCurrentUser] = useState(() => {
    try {
      const stored = localStorage.getItem('pennydime_auth_user');
      return stored ? JSON.parse(stored) : null;
    } catch {
      return null;
    }
  });

  // Current Route Hash Navigation
  const [currentRoute, setCurrentRouteState] = useState(() => {
    const hash = window.location.hash.replace('#', '');
    if (VALID_ROUTES.includes(hash)) {
      return hash;
    }
    return 'dashboard';
  });

  const [isMobileMenuOpen, setIsMobileMenuOpen] = useState(false);

  const setCurrentRoute = (route) => {
    setCurrentRouteState(route);
    window.location.hash = route;
    setIsMobileMenuOpen(false);
    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  useEffect(() => {
    const handleHashChange = () => {
      const hash = window.location.hash.replace('#', '');
      if (VALID_ROUTES.includes(hash)) {
        setCurrentRouteState(hash);
      }
    };
    window.addEventListener('hashchange', handleHashChange);
    return () => window.removeEventListener('hashchange', handleHashChange);
  }, []);

  // 2. User-Scoped Ledger Data State
  const [period, setPeriod] = useState('Month');
  const [selectedMonth, setSelectedMonth] = useState(() => {
    return new Date().toLocaleDateString('en-US', { month: 'long', year: 'numeric' });
  });
  const [searchQuery, setSearchQuery] = useState('');
  const [startingBalance, setStartingBalance] = useState(0);
  const [currencySymbol, setCurrencySymbol] = useState('₹');
  const [userProfile, setUserProfile] = useState(() => ({
    name: currentUser?.name || 'Ledger User',
    email: currentUser?.email || '',
    tier: currentUser?.tier || 'Private Ledger Member',
    avatar: currentUser?.avatar || ''
  }));
  const [allowancePool, setAllowancePool] = useState(initialRealData.allowancePool);
  const [budgetGoals, setBudgetGoals] = useState(initialRealData.budgetGoals);
  const [upcomingAdjustments, setUpcomingAdjustments] = useState([]);
  const [transactions, setTransactions] = useState([]);

  // Modals & UI State
  const [isAddEntryOpen, setIsAddEntryOpen] = useState(false);
  const [isSetStartingOpen, setIsSetStartingOpen] = useState(false);
  const [isReconcileOpen, setIsReconcileOpen] = useState(false);
  const [isAdjustPoolOpen, setIsAdjustPoolOpen] = useState(false);
  const [activeAdjustBudget, setActiveAdjustBudget] = useState(null);
  const [toasts, setToasts] = useState([]);

  const showToast = (message) => {
    const id = Date.now();
    setToasts((prev) => [...prev, { id, message }]);
    setTimeout(() => {
      setToasts((prev) => prev.filter((t) => t.id !== id));
    }, 3500);
  };

  // Load User Data from localStorage upon login
  useEffect(() => {
    if (!currentUser?.email) return;

    try {
      const storageKey = `pennydime_ledger_${currentUser.email.toLowerCase()}`;
      const savedData = localStorage.getItem(storageKey);

      if (savedData) {
        const parsed = JSON.parse(savedData);
        setStartingBalance(parsed.startingBalance ?? 0);
        setCurrencySymbol(parsed.currencySymbol || '₹');
        setUserProfile({
          name: currentUser.name || parsed.userProfile?.name || 'Ledger User',
          email: currentUser.email || parsed.userProfile?.email || '',
          tier: currentUser.tier || parsed.userProfile?.tier || 'Verified Sovereign Member',
          avatar: currentUser.avatar || parsed.userProfile?.avatar || ''
        });
        if (parsed.allowancePool) setAllowancePool(parsed.allowancePool);
        if (parsed.budgetGoals) setBudgetGoals(parsed.budgetGoals);
        if (parsed.upcomingAdjustments) setUpcomingAdjustments(parsed.upcomingAdjustments);
        if (Array.isArray(parsed.transactions)) setTransactions(parsed.transactions);
      } else {
        // Initialize fresh real ledger for brand new user
        setStartingBalance(0);
        setCurrencySymbol('₹');
        setUserProfile({
          name: currentUser.name || 'Ledger User',
          email: currentUser.email || '',
          tier: currentUser.tier || 'Private Ledger Member',
          avatar: currentUser.avatar || ''
        });
        setTransactions([]);
        setBudgetGoals(defaultCategories.map((c) => ({ ...c, spent: 0 })));
        setAllowancePool({
          ...initialRealData.allowancePool,
          spent: 0
        });
      }
    } catch (err) {
      console.error('Failed to parse user ledger data', err);
    }
  }, [currentUser]);

  // Persist User Data to localStorage on change
  useEffect(() => {
    if (!currentUser?.email) return;

    try {
      const storageKey = `pennydime_ledger_${currentUser.email.toLowerCase()}`;
      const payload = {
        startingBalance,
        currencySymbol,
        userProfile,
        allowancePool,
        budgetGoals,
        upcomingAdjustments,
        transactions,
        lastSaved: new Date().toISOString()
      };
      localStorage.setItem(storageKey, JSON.stringify(payload));
    } catch (err) {
      console.error('Failed to save ledger data', err);
    }
  }, [currentUser, startingBalance, currencySymbol, userProfile, allowancePool, budgetGoals, upcomingAdjustments, transactions]);

  // Authentication Handlers
  const handleLoginSuccess = (user) => {
    localStorage.setItem('pennydime_auth_user', JSON.stringify(user));
    setCurrentUser(user);
    showToast(`Welcome back, ${user.name}!`);
  };

  const handleLogout = () => {
    localStorage.removeItem('pennydime_auth_user');
    setCurrentUser(null);
    setCurrentRoute('dashboard');
    showToast('You have been securely signed out.');
  };

  // Dynamic Real-time Calculations
  const crTransactions = useMemo(
    () => transactions.filter((t) => t.type === 'CR'),
    [transactions]
  );
  const drTransactions = useMemo(
    () => transactions.filter((t) => t.type === 'DR'),
    [transactions]
  );
  const totalCR = useMemo(
    () => crTransactions.reduce((acc, curr) => acc + (parseFloat(curr.amount) || 0), 0),
    [crTransactions]
  );
  const totalDR = useMemo(
    () => drTransactions.reduce((acc, curr) => acc + (parseFloat(curr.amount) || 0), 0),
    [drTransactions]
  );
  const currentCalculatedBalance = startingBalance + totalCR - totalDR;

  // Real-time computed budget goals reflecting actual DR transactions
  const dynamicallyComputedBudgets = useMemo(() => {
    return budgetGoals.map((b) => {
      const actualSpent = drTransactions
        .filter((t) => t.category.toLowerCase() === b.category.toLowerCase())
        .reduce((sum, t) => sum + (parseFloat(t.amount) || 0), 0);
      return {
        ...b,
        spent: actualSpent
      };
    });
  }, [budgetGoals, drTransactions]);

  // Real-time computed allowance pool reflecting actual transactions
  const dynamicallyComputedPool = useMemo(() => {
    const actualPoolSpent = drTransactions
      .filter(
        (t) =>
          t.allowancePool === 'travel' ||
          t.category.toLowerCase() === allowancePool.category.toLowerCase()
      )
      .reduce((sum, t) => sum + (parseFloat(t.amount) || 0), 0);
    return {
      ...allowancePool,
      spent: actualPoolSpent
    };
  }, [allowancePool, drTransactions]);

  // Add Transaction Entry
  const handleAddEntry = (newEntry) => {
    const isCR = newEntry.type === 'CR';
    const firstTx = transactions[0];
    const prevBal = firstTx ? firstTx.runningBal : startingBalance;
    const newRunningBal = isCR ? prevBal + newEntry.amount : prevBal - newEntry.amount;

    const entryWithBal = {
      ...newEntry,
      runningBal: newRunningBal
    };

    setTransactions((prev) => [entryWithBal, ...prev]);

    showToast(
      `Recorded ${newEntry.type}: ${newEntry.title} (${currencySymbol}${newEntry.amount.toLocaleString('en-IN')})`
    );
  };

  // Toggle Reconciled
  const handleToggleReconciled = (txId) => {
    setTransactions((prev) =>
      prev.map((t) => {
        if (t.id === txId) {
          const nextState = !t.isReconciled;
          showToast(nextState ? `Reconciled: ${t.title}` : `Marked uncertified: ${t.title}`);
          return { ...t, isReconciled: nextState };
        }
        return t;
      })
    );
  };

  // Delete Transaction
  const handleDeleteTransaction = (txId) => {
    setTransactions((prev) => prev.filter((t) => t.id !== txId));
    showToast('Transaction entry deleted');
  };

  // Accept Adjustment
  const handleAcceptAdjustment = (adjId) => {
    setUpcomingAdjustments((prev) =>
      prev.map((adj) => {
        if (adj.id === adjId) {
          const nextState = !adj.accepted;
          if (nextState) {
            showToast(
              `Adjustment accepted for ${adj.name}! Allocation updated to ${adj.currency || currencySymbol}${adj.newVal}`
            );
          }
          return { ...adj, accepted: nextState };
        }
        return adj;
      })
    );
  };

  const handleReconcileCategory = (catName) => {
    showToast(`Category "${catName}" reconciled against statement`);
  };

  // Reset or Load Sample Data
  const handleResetData = () => {
    setStartingBalance(0);
    setTransactions([]);
    setBudgetGoals(defaultCategories.map((c) => ({ ...c, spent: 0 })));
    setAllowancePool({
      ...initialRealData.allowancePool,
      spent: 0
    });
    setUpcomingAdjustments([]);
    showToast('Real ledger cleared to pristine zero state');
  };

  const handleLoadSampleDemo = () => {
    setStartingBalance(20000);
    setTransactions(sampleDemoTransactions);
    showToast('Optional sample entries loaded for preview');
  };

  // UNLESS USER HAS LOGGED IN, USER CANNOT ACCESS THE SITE
  if (!currentUser) {
    return <AuthPage onLoginSuccess={handleLoginSuccess} />;
  }

  return (
    <div className="app-layout">
      {/* Desktop & Mobile Responsive Sidebar */}
      <Sidebar
        currentRoute={currentRoute}
        setCurrentRoute={setCurrentRoute}
        userProfile={userProfile}
        isMobileOpen={isMobileMenuOpen}
        onCloseMobile={() => setIsMobileMenuOpen(false)}
        onLogout={handleLogout}
      />

      <main className="main-content">
        <Header
          period={period}
          setPeriod={setPeriod}
          selectedMonth={selectedMonth}
          setSelectedMonth={setSelectedMonth}
          searchQuery={searchQuery}
          setSearchQuery={setSearchQuery}
          onOpenAddEntry={() => setIsAddEntryOpen(true)}
          currentRoute={currentRoute}
          onToggleMobileMenu={() => setIsMobileMenuOpen(!isMobileMenuOpen)}
          userProfile={userProfile}
        />

        <div className="main-page-body">
          <AppRoutes
            currentRoute={currentRoute}
            startingBalance={startingBalance}
            transactions={transactions}
            dynamicallyComputedPool={dynamicallyComputedPool}
            currencySymbol={currencySymbol}
            onOpenSetStarting={() => setIsSetStartingOpen(true)}
            onOpenReconcile={() => setIsReconcileOpen(true)}
            onOpenAdjustPool={() => setIsAdjustPoolOpen(true)}
            onNavigateToTransactions={() => setCurrentRoute('transactions')}
            onOpenAddEntry={() => setIsAddEntryOpen(true)}
            searchQuery={searchQuery}
            onToggleReconciled={handleToggleReconciled}
            onDeleteTransaction={handleDeleteTransaction}
            dynamicallyComputedBudgets={dynamicallyComputedBudgets}
            upcomingAdjustments={upcomingAdjustments}
            onOpenAdjustBudget={(budget) => setActiveAdjustBudget(budget)}
            onAcceptAdjustment={handleAcceptAdjustment}
            onReconcileCategory={handleReconcileCategory}
            onAddAllowancePool={() => setIsAdjustPoolOpen(true)}
            userProfile={userProfile}
            setUserProfile={setUserProfile}
            setCurrencySymbol={setCurrencySymbol}
            setStartingBalance={setStartingBalance}
            onResetData={handleResetData}
            onLoadSampleDemo={handleLoadSampleDemo}
            onLogout={handleLogout}
            showToast={showToast}
          />
        </div>
      </main>

      {/* Common Mobile Bottom Navigation & Floating Action Button */}
      <MobileNav
        currentRoute={currentRoute}
        setCurrentRoute={setCurrentRoute}
        onOpenAddEntry={() => setIsAddEntryOpen(true)}
      />

      {/* Common Add Entry Modal */}
      <AddEntryModal
        isOpen={isAddEntryOpen}
        onClose={() => setIsAddEntryOpen(false)}
        onAddEntry={handleAddEntry}
        currencySymbol={currencySymbol}
        allowancePool={dynamicallyComputedPool}
      />

      {/* Set Starting Balance Modal */}
      <SetStartingModal
        isOpen={isSetStartingOpen}
        onClose={() => setIsSetStartingOpen(false)}
        startingBalance={startingBalance}
        onSave={(newBal) => {
          setStartingBalance(newBal);
          showToast(
            `Baseline starting balance updated to ${currencySymbol}${newBal.toLocaleString('en-IN')}`
          );
        }}
        currencySymbol={currencySymbol}
      />

      {/* Reconcile Modal */}
      <ReconcileModal
        isOpen={isReconcileOpen}
        onClose={() => setIsReconcileOpen(false)}
        currentBalance={currentCalculatedBalance}
        onReconciled={() => {
          showToast('Ledger successfully reconciled and certified!');
        }}
        currencySymbol={currencySymbol}
      />

      {/* Adjust Allowance Pool Modal */}
      <AdjustPoolModal
        isOpen={isAdjustPoolOpen}
        onClose={() => setIsAdjustPoolOpen(false)}
        pool={dynamicallyComputedPool}
        onSave={(updatedPool) => {
          setAllowancePool(updatedPool);
          showToast(`Allowance pool updated: ${updatedPool.name}`);
        }}
        currencySymbol={currencySymbol}
      />

      {/* Adjust Budget Modal */}
      <AdjustBudgetModal
        isOpen={activeAdjustBudget !== null}
        onClose={() => setActiveAdjustBudget(null)}
        budget={activeAdjustBudget}
        onSave={(updatedBudget) => {
          setBudgetGoals((prev) =>
            prev.map((b) => (b.id === updatedBudget.id ? updatedBudget : b))
          );
          showToast(`Budget limit updated for ${updatedBudget.category}`);
        }}
        currencySymbol={currencySymbol}
      />

      {/* Toast Notification Container */}
      <div className="toast-container">
        {toasts.map((t) => (
          <div key={t.id} className="toast-item">
            <span>{t.message}</span>
          </div>
        ))}
      </div>
    </div>
  );
}

import React, { useState, useEffect } from 'react';
import { X, Sliders } from 'lucide-react';

export default function AdjustBudgetModal({
  isOpen,
  onClose,
  budget,
  onSave,
  currencySymbol = '₹'
}) {
  const [allocated, setAllocated] = useState(budget ? budget.allocated : 0);
  const [spent, setSpent] = useState(budget ? budget.spent : 0);

  useEffect(() => {
    if (budget) {
      setAllocated(budget.allocated);
      setSpent(budget.spent);
    }
  }, [budget]);

  if (!isOpen || !budget) return null;

  const handleSubmit = (e) => {
    e.preventDefault();
    onSave({
      ...budget,
      allocated: parseFloat(allocated) || 0,
      spent: parseFloat(spent) || 0
    });
    onClose();
  };

  return (
    <div className="modal-backdrop" onClick={onClose}>
      <div className="modal-box" onClick={(e) => e.stopPropagation()}>
        <div className="modal-header">
          <h2 className="modal-title" style={{ display: 'flex', alignItems: 'center', gap: '8px' }}>
            <Sliders size={20} color="#1b3a2f" />
            <span>Adjust Limit: {budget.category}</span>
          </h2>
          <button className="modal-close-btn" onClick={onClose} aria-label="Close">
            <X size={20} />
          </button>
        </div>

        <form onSubmit={handleSubmit}>
          <div className="modal-body">
            <div className="settings-field-group">
              <label className="settings-label">
                MONTHLY LIMIT ALLOCATION ({currencySymbol})
              </label>
              <input
                type="number"
                step="any"

const PALETTE = ['#e8703a', '#03241a', '#1b3a2f', '#456558', '#83a496', '#fd8048', '#727974'];

export default function DebitAllocationDonut({ transactions = [], totalDR = 0, currencySymbol = '₹' }) {
  const drTransactions = transactions.filter((t) => t.type === 'DR');

  // Dynamically group debits by category
  const categoryMap = {};
  drTransactions.forEach((tx) => {
    const cat = tx.category || 'Others';
    categoryMap[cat] = (categoryMap[cat] || 0) + (parseFloat(tx.amount) || 0);
  });

  const categoryEntries = Object.entries(categoryMap).sort((a, b) => b[1] - a[1]);
  const calculatedTotalDR = categoryEntries.reduce((sum, [, amt]) => sum + amt, 0);

  const categories = categoryEntries.map(([name, amount], index) => {
    const pctVal = calculatedTotalDR > 0 ? (amount / calculatedTotalDR) * 100 : 0;
    return {
      name,
      amount,
      pctVal,
      pct: `${pctVal.toFixed(1)}%`,
      color: PALETTE[index % PALETTE.length]
    };
  });

  const circumference = 2 * Math.PI * 68; // ~427.25

  let accumulatedOffset = 0;
  const svgSegments = categories.map((cat) => {
    const segmentLength = (cat.pctVal / 100) * circumference;
    const strokeDasharray = `${segmentLength} ${circumference}`;
    const strokeDashoffset = -accumulatedOffset;
    accumulatedOffset += segmentLength;

    return {
      name: cat.name,
      color: cat.color,
      strokeDasharray,
      strokeDashoffset
    };
  });

  return (
    <div className="card-white debit-allocation-card">
      <div className="card-inner-header">
        <div>
          <h2 className="card-title-serif">Debit Allocation</h2>

import React, { useState } from 'react';
import { Plus } from 'lucide-react';
import TransactionMetrics from '../../components/transactions/TransactionMetrics';
import TransactionFilters from '../../components/transactions/TransactionFilters';
import TransactionGroupList from '../../components/transactions/TransactionGroupList';

export default function TransactionsPage({
  transactions,
  currencySymbol = '₹',
  searchQuery,
  onOpenAddEntry,
  onToggleReconciled
}) {
  const [filterType, setFilterType] = useState('ALL');
  const [sortOrder, setSortOrder] = useState('DATE_DESC');

  const getFilteredTransactions = () => {
    let list = [...transactions];

    if (searchQuery.trim()) {
      const q = searchQuery.toLowerCase();
      list = list.filter(
        (t) =>
          t.title.toLowerCase().includes(q) ||
          t.category.toLowerCase().includes(q) ||
          (t.note && t.note.toLowerCase().includes(q))
      );
    }

    if (filterType === 'DR_ONLY') {
      list = list.filter((t) => t.type === 'DR');
    } else if (filterType === 'CR_ONLY') {
      list = list.filter((t) => t.type === 'CR');
    } else if (filterType === 'RECONCILED') {
      list = list.filter((t) => t.isReconciled);
    }

    list.sort((a, b) => {
      if (sortOrder === 'DATE_DESC') return new Date(b.date) - new Date(a.date);
      if (sortOrder === 'DATE_ASC') return new Date(a.date) - new Date(b.date);
      if (sortOrder === 'AMOUNT_DESC') return b.amount - a.amount;
      if (sortOrder === 'AMOUNT_ASC') return a.amount - b.amount;
      return 0;
    });

    return list;
  };

  const filtered = getFilteredTransactions();

  const totalCR = transactions
    .filter((t) => t.type === 'CR')
    .reduce((sum, t) => sum + t.amount, 0);

  const totalDR = transactions
    .filter((t) => t.type === 'DR')
    .reduce((sum, t) => sum + t.amount, 0);

  return (
    <div className="transactions-page-wrap">
      <div className="page-intro-header">
        <div>
          <h1 className="page-intro-title">Transaction Ledger</h1>
          <p className="page-intro-subtitle">
            Audited history of incoming and outgoing flows with running balances and reconciliation markers.
          </p>
        </div>

        <button
          type="button"
          className="btn-primary"
          style={{
            backgroundColor: '#03241a',
            display: 'flex',
            alignItems: 'center',
            gap: '8px',
            padding: '10px 18px',
            borderRadius: '9999px'
          }}
          onClick={onOpenAddEntry}
        >
          <Plus size={16} />
          <span>Add Entry</span>
        </button>
      </div>

      {/* Summary strip: CR, DR, Net */}
      <TransactionMetrics
        totalCR={totalCR}
        totalDR={totalDR}
        currencySymbol={currencySymbol}
      />

      {/* Controls: Filter & Sort */}
      <TransactionFilters
        filterType={filterType}
        setFilterType={setFilterType}
        sortOrder={sortOrder}
        setSortOrder={setSortOrder}
        filteredCount={filtered.length}
      />

      {/* Grouped Day Lists */}
      <TransactionGroupList
        filteredTransactions={filtered}
        currencySymbol={currencySymbol}
        onToggleReconciled={onToggleReconciled}
      />
    </div>
  );
}

/**
 * Formatting utilities for PennyDime Ledger
 */

export const formatCurrency = (val, symbol = '₹') => {
  const num = Number(val) || 0;
  return (
    symbol +
    num.toLocaleString('en-IN', {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2,
    })
  );
};

export const formatCurrencyInteger = (val, symbol = '₹') => {
  const num = Math.round(Number(val) || 0);
  return symbol + num.toLocaleString('en-IN');
};

export const formatDateDisplay = (dateStr) => {
  if (!dateStr) return 'Today';
  if (dateStr.includes('Mar') || dateStr.includes('Oct') || dateStr.includes('2026')) {
    return dateStr;
  }
  try {
    const d = new Date(dateStr);
    return d.toLocaleDateString('en-US', {
      day: 'numeric',
      month: 'short',
      year: 'numeric',
    });
  } catch {
    return dateStr;
  }
};
